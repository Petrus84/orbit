# Cores
$GREEN = 'Green'
$RED = 'Red'
$YELLOW = 'Yellow'
$CYAN = 'Cyan'

# ========== FUNÇÃO: Verificar Arquivo ==========
function Test-FileExists {
    param(
        [string]$FilePath,
        [string]$Description
    )
    
    if (Test-Path $FilePath) {
        Write-Host "  ✓ $Description" -ForegroundColor $GREEN
        return $true
    } else {
        Write-Host "  ✗ $Description (NÃO ENCONTRADO)" -ForegroundColor $RED
        return $false
    }
}

# ========== FUNÇÃO: Verificar Importação ==========
function Test-ImportExists {
    param(
        [string]$FilePath,
        [string]$ImportName,
        [string]$Description
    )
    
    if (-not (Test-Path $FilePath)) {
        Write-Host "  ✗ $Description (arquivo não existe)" -ForegroundColor $RED
        return $false
    }
    
    $content = Get-Content $FilePath -Raw
    if ($content -match $ImportName) {
        Write-Host "  ✓ $Description" -ForegroundColor $GREEN
        return $true
    } else {
        Write-Host "  ✗ $Description (importação não encontrada)" -ForegroundColor $RED
        return $false
    }
}

# ========== FUNÇÃO: Verificar Função Exportada ==========
function Test-ExportedFunction {
    param(
        [string]$FilePath,
        [string]$FunctionName,
        [string]$Description
    )
    
    if (-not (Test-Path $FilePath)) {
        Write-Host "  ✗ $Description (arquivo não existe)" -ForegroundColor $RED
        return $false
    }
    
    $content = Get-Content $FilePath -Raw
    if ($content -match "export.*function $FunctionName") {
        Write-Host "  ✓ $Description" -ForegroundColor $GREEN
        return $true
    } else {
        Write-Host "  ✗ $Description (função não exportada)" -ForegroundColor $RED
        return $false
    }
}

# ========== FUNÇÃO: Contar Ocorrências ==========
function Get-OccurrenceCount {
    param(
        [string]$Pattern,
        [string]$FilePath
    )
    
    if (-not (Test-Path $FilePath)) {
        return 0
    }
    
    $content = Get-Content $FilePath -Raw
    $matches = [regex]::Matches($content, $Pattern)
    return $matches.Count
}
